  const showReport = Boolean(evaluationReport);
  const showConfig = !isActive && !showReport;

  return (
    <main className="flex min-h-screen flex-col items-center bg-slate-50 p-4 sm:p-8 text-slate-800 font-sans">
      <div className="mb-6 flex w-full max-w-[1400px] justify-end">
        {showReport && (
          <button
            onClick={handleStartAnotherInterview}
            className="rounded-full bg-blue-600 px-5 py-2.5 font-medium text-white shadow-lg shadow-blue-600/20 transition-all hover:bg-blue-700"
          >
            Start another interview
          </button>
        )}
      </div>

      {isActive && !evaluationReport && (
        <div className="mb-6 w-full max-w-6xl">
          <div
            className={`
              relative rounded-2xl border px-6 py-4 transition-all duration-500
              ${isCritical
                ? 'border-red-300 bg-red-50 shadow-lg shadow-red-100/60'
                : isWarning
                  ? 'border-amber-300 bg-amber-50 shadow-md shadow-amber-100/40'
                  : 'border-gray-200 bg-white shadow-sm'}
            `}
          >
            <div className="pointer-events-none absolute inset-0 overflow-hidden rounded-2xl">
              <div
                className={`h-full opacity-10 transition-all duration-1000 ease-linear ${
                  isCritical ? 'bg-red-500' : isWarning ? 'bg-amber-500' : 'bg-blue-500'
                }`}
                style={{ width: `${timerProgress * 100}%` }}
              />
            </div>

            <div className="relative flex items-center justify-between">
              <div className="flex items-center gap-3">
                <div
                  className={`
                    h-3 w-3 rounded-full
                    ${isCritical
                      ? 'bg-red-500 animate-ping'
                      : isWarning
                        ? 'bg-amber-500 animate-pulse'
                        : 'bg-emerald-500'}
                  `}
                />
                <span
                  className={`text-sm font-semibold uppercase tracking-wider ${
                    isCritical
                      ? 'text-red-600'
                      : isWarning
                        ? 'text-amber-600'
                        : 'text-gray-400'
                  }`}
                >
                  {isCritical
                    ? 'Wrap up your answer'
                    : isWarning
                      ? 'Last 30 seconds'
                      : isEnding && !closingReceived
                        ? 'Interviewer wrapping up...'
                        : isEnding && closingReceived
                          ? 'Generating evaluation...'
                          : 'Interview in progress'}
                </span>
              </div>

              <div
                className={`
                  font-mono text-2xl font-bold tracking-wider tabular-nums
                  ${isCritical
                    ? 'animate-pulse text-red-600'
                    : isWarning
                      ? 'text-amber-600'
                      : 'text-slate-700'}
                `}
              >
                {formatTime(timeRemaining)}
              </div>
            </div>

            {isCritical && timeRemaining > 0 && (
              <div className="relative mt-2 text-center">
                <span className="text-xs font-semibold tracking-wide text-red-500">
                  {timeRemaining} second{timeRemaining !== 1 ? 's' : ''} remaining - finish your response
                </span>
              </div>
            )}
          </div>
        </div>
      )}

      <div
        className={`
          w-full gap-6
          ${showReport
            ? 'grid max-w-[1400px] lg:grid-cols-[minmax(420px,1fr)_420px]'
            : showConfig
              ? 'flex flex-col items-center justify-center max-w-4xl mx-auto mt-10'
              : 'grid max-w-4xl mx-auto items-start'}
        `}
      >
        {showConfig && (
          <div className="w-full">
            <InterviewConfig 
              onChange={setInterviewConfig} 
              onFinish={() => {
                setIsConfigFinished(true);
                startInterview();
              }}
              disabled={isActive || showReport} 
            />
          </div>
        )}

        {!showConfig && (
          <div className="min-w-0 w-full">
            <VideoCapture isActive={isActive} onFrame={handleFrame} analysis={analysis} />

          {isActive && !evaluationReport && (
            <div className="mt-4 w-full">
              <div className="rounded-2xl border border-slate-200 bg-slate-100/50 shadow-sm">
                {/* Chat header */}
                <div className="flex items-center gap-2 px-5 py-3 border-b border-slate-200/60">
                  <MessageSquare className="h-4 w-4 text-slate-400" />
                  <span className="text-xs font-semibold uppercase tracking-wider text-slate-400">
                    Conversation
                  </span>
                </div>

                {/* Scrollable chat area */}
                <div
                  ref={chatScrollRef}
                  className="flex flex-col gap-3 overflow-y-auto max-h-[380px] px-4 py-4 scroll-smooth"
                >
                  {messages.length === 0 && !currentAiMessage && (
                    <div className="text-center py-10">
                      <Bot className="mx-auto h-8 w-8 text-slate-200 mb-2" />
                      <p className="text-sm text-slate-400">The interview conversation will appear here</p>
                    </div>
                  )}

                  {/* Message history */}
                  {messages.map((msg, index) => (
                    <div
                      key={index}
                      className={`flex gap-2.5 ${msg.role === 'user' ? 'flex-row-reverse' : 'flex-row'}`}
                    >
                      <div className={`
                        flex h-7 w-7 shrink-0 items-center justify-center rounded-full mt-0.5
                        ${msg.role === 'user'
                          ? 'bg-blue-100 text-blue-600'
                          : 'bg-indigo-100 text-indigo-600'}
                      `}>
                        {msg.role === 'user'
                          ? <User className="h-3.5 w-3.5" />
                          : <Bot className="h-3.5 w-3.5" />
                        }
                      </div>
                      <div className={`
                        max-w-[80%] rounded-2xl px-4 py-2.5 text-sm leading-relaxed
                        ${msg.role === 'user'
                          ? 'bg-blue-600 text-white rounded-tr-md'
                          : 'bg-white border border-slate-100 text-slate-700 shadow-sm rounded-tl-md'}
                      `}>
                        {msg.text}
                      </div>
                    </div>
                  ))}

                  {/* Streaming AI message (in-progress) */}
                  {currentAiMessage && (
                    <div className="flex gap-2.5 flex-row">
                      <div className="flex h-7 w-7 shrink-0 items-center justify-center rounded-full mt-0.5 bg-indigo-100 text-indigo-600">
                        <Bot className="h-3.5 w-3.5" />
                      </div>
                      <div className="max-w-[80%] rounded-2xl rounded-tl-md px-4 py-2.5 text-sm leading-relaxed bg-white border border-indigo-100 text-slate-700 shadow-sm">
                        {currentAiMessage}
                        <span className="ml-0.5 inline-block h-4 w-0.5 animate-pulse bg-indigo-500 align-text-bottom" />
                      </div>
                    </div>
                  )}

                  {/* Typing indicator (waiting for first token) */}
                  {isAiSpeaking && !currentAiMessage && (
                    <div className="flex gap-2.5 flex-row">
                      <div className="flex h-7 w-7 shrink-0 items-center justify-center rounded-full mt-0.5 bg-indigo-100 text-indigo-600">
                        <Bot className="h-3.5 w-3.5" />
                      </div>
                      <div className="rounded-2xl rounded-tl-md px-4 py-3 bg-white border border-slate-100 shadow-sm">
                        <div className="flex gap-1">
                          <div className="h-2 w-2 rounded-full bg-indigo-300 animate-bounce" style={{ animationDelay: '0ms' }} />
                          <div className="h-2 w-2 rounded-full bg-indigo-300 animate-bounce" style={{ animationDelay: '150ms' }} />
                          <div className="h-2 w-2 rounded-full bg-indigo-300 animate-bounce" style={{ animationDelay: '300ms' }} />
                        </div>
                      </div>
                    </div>
                  )}
                </div>
              </div>
            </div>
          )}

          {isActive && !evaluationReport && (
            <div className="mt-4">
              <CodeEditor
                subject={interviewConfig.subject}
                onCodeChange={(code) => { codeContentRef.current = code; }}
                collapsed={true}
              />
            </div>
          )}

          {!evaluationReport && (
            <div className="mt-4 flex h-10 w-full justify-center">
              <AudioStreamer
                ref={audioStreamerRef}
                isActive={isActive}
                config={interviewConfig}
                onEvaluation={handleEvaluation}
                onAiToken={handleAiToken}
                onAiTurnEnd={handleAiTurnEnd}
                onSessionId={handleSessionId}
                onTranscriptReady={handleTranscriptReady}
                onUserTranscript={handleUserTranscript}
              />
            </div>
          )}

          {!evaluationReport && (
            <div className="mt-4 flex items-center justify-center gap-4">
              {isActive && !closingReceived && (
              <button
                  onClick={handleDoneClick}
                  disabled={!hasTranscript || isAiSpeaking}
                  className={`z-10 rounded-full px-8 py-3 font-medium shadow-xl transition-all ${
                    hasTranscript && !isAiSpeaking
                      ? 'bg-emerald-600 text-white shadow-emerald-600/20 hover:bg-emerald-700 cursor-pointer'
                      : 'bg-slate-200 text-slate-400 shadow-none cursor-not-allowed'
                  }`}
                >
                  {isTimerZero && !isEnding ? 'Send final answer' : 'Done'}
                </button>
              )}

              {!isActive && isConfigFinished && (
                <button
                  onClick={startInterview}
                  className="z-10 rounded-full bg-blue-600 px-8 py-3 font-medium text-white shadow-xl shadow-blue-600/20 transition-all hover:bg-blue-700"
                >
                  Start Interview
                </button>
              )}

              {isActive && isEnding && !closingReceived && (
                <div className="rounded-full bg-blue-100 px-8 py-3 font-medium text-blue-500 animate-pulse">
                  Interviewer wrapping up...
                </div>
              )}
              {isActive && isEnding && closingReceived && (
                <div className="rounded-full bg-gray-200 px-8 py-3 font-medium text-gray-400">
                  Generating evaluation...
                </div>
              )}
            </div>
          )}
        </div>
        )}

        {showReport && (
          <div className="min-w-0">
            <EvaluationReport report={evaluationReport} compact />
          </div>
        )}
      </div>
    </main>
  );
}
